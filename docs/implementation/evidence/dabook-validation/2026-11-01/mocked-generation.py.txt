from __future__ import annotations

import hashlib
import json
import re
import sys
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

PORT = int(sys.argv[1])
LOG = Path(sys.argv[2])


class Handler(BaseHTTPRequestHandler):
    def do_POST(self) -> None:  # noqa: N802
        length = int(self.headers.get("Content-Length", "0"))
        request = json.loads(self.rfile.read(length))
        user = request["messages"][-1]["content"]
        match = re.search(r"<<<EXCERPT\s*(.*?)\s*EXCERPT>>>", user, re.DOTALL)
        excerpt = match.group(1).strip() if match else ""
        sentences = [s.strip() for s in re.split(r"(?<=[.!?])\s+", excerpt) if s.strip()]
        evidence = next((s for s in sentences if len(s.split()) >= 15), "")
        section = re.search(r"Section\s+(\d+)", evidence)
        number = section.group(1) if section else "unknown"
        content = json.dumps(
            {
                "pairs": [
                    {
                        "category": "explain",
                        "question": f"What does Section {number} explain about this bounded rehearsal?",
                        "answer": evidence,
                        "evidence": evidence,
                    }
                ]
            }
        )
        response = {
            "id": "codex-authored-local",
            "object": "chat.completion",
            "model": "codex-authored-local",
            "choices": [{"message": {"role": "assistant", "content": content}}],
            "usage": {
                "prompt_tokens": len(user.split()),
                "completion_tokens": len(content.split()),
            },
        }
        body = json.dumps(response).encode()
        record = {
            "path": self.path,
            "excerpt_sha256": hashlib.sha256(excerpt.encode()).hexdigest(),
            "response_sha256": hashlib.sha256(content.encode()).hexdigest(),
            "evidence": evidence,
        }
        with LOG.open("a", encoding="utf-8") as stream:
            stream.write(json.dumps(record) + "\n")
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *_args: object) -> None:
        return


ThreadingHTTPServer(("127.0.0.1", PORT), Handler).serve_forever()
